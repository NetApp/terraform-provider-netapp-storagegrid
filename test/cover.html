// decode converts a decoded JSON object into a struct, or a struct into a map, using the mapstructure tags.
// Types are converted when it is safe: the REST client keeps numbers as json.Number, and StorageGRID is not always
// consistent between strings and numbers (eg account ids).
func decode(input interface{}, output interface{}) error {
        decoder, err := mapstructure.NewDecoder(&mapstructure.DecoderConfig{
                WeaklyTypedInput: true,
                Result:           output,
        })
        if err != nil {
                return err
        }
        return decoder.Decode(input)
}

// gridAccountsAPI is the Grid Management API of the tenant accounts, relative to the version of the API.
const gridAccountsAPI = "grid/accounts"

var gridAccountIDRegexp = regexp.MustCompile(`^[0-9]+$`)

// IsGridAccountID returns true if id looks like the id that StorageGRID assigns to a tenant account: a number of 20
// digits.  The id is a part of the path of the requests, so anything else is not sent: a path like ../users/1 would
// address another object.
func IsGridAccountID(id string) bool {
        return gridAccountIDRegexp.MatchString(id)
}

// gridAccountPath returns the path of an account, and reports an error if id is not the id of an account.
func gridAccountPath(errorHandler *utils.ErrorHandler, id string, suffix string) (string, error) {
        if !IsGridAccountID(id) {
                return "", errorHandler.MakeAndReportError("invalid grid_account id", fmt.Sprintf("%q is not the id of a grid_account, which is a number such as 12345678901234567890", id))
        }
        return gridAccountsAPI + "/" + id + suffix, nil
}

// GridAccountGetDataModel describes a tenant account returned by StorageGRID.
// StorageGRID leaves out the fields that are not set, or sets them to null: they are nil.
type GridAccountGetDataModel struct {
        ID           string                        `mapstructure:"id"`
        Name         string                        `mapstructure:"name"`
        Description  *string                       `mapstructure:"description"`
        Capabilities []string                      `mapstructure:"capabilities"`
        Policy       GridAccountPolicyGetDataModel `mapstructure:"policy"`
}

// GridAccountPolicyGetDataModel describes the policy of a tenant account returned by StorageGRID.
type GridAccountPolicyGetDataModel struct {
        UseAccountIdentitySource *bool  `mapstructure:"useAccountIdentitySource"`
        AllowPlatformServices    *bool  `mapstructure:"allowPlatformServices"`
        AllowSelectObjectContent *bool  `mapstructure:"allowSelectObjectContent"`
        AllowComplianceMode      *bool  `mapstructure:"allowComplianceMode"`
        MaxRetentionDays         *int64 `mapstructure:"maxRetentionDays"`
        QuotaObjectBytes         *int64 `mapstructure:"quotaObjectBytes"`

// GridAccountResourceBodyDataModel describes the configuration of a tenant account that the provider sends.
type GridAccountResourceBodyDataModel struct {
        Name         string
        Description  *string
        Capabilities []string
        Policy       GridAccountPolicyBodyDataModel
        // Password is the password of the root user of the tenant.  It is only sent when the account is created:
        // ChangeGridAccountPassword changes it afterwards.
        Password string
        // GrantRootAccessToGroup is the federated group that has the root access permission of the tenant.
        // It is only sent when the account is created.
        GrantRootAccessToGroup string
}

// GridAccountPolicyBodyDataModel describes the policy of a tenant account that the provider sends.
// The pointers are nil when the provider does not manage the setting: it is not sent on create, and the
// value that StorageGRID has is sent back on update.  Only QuotaObjectBytes is always sent, nil is no quota.
type GridAccountPolicyBodyDataModel struct {
        UseAccountIdentitySource bool
        AllowPlatformServices    bool
        AllowSelectObjectContent *bool
        AllowComplianceMode      *bool
        MaxRetentionDays         *int64
        QuotaObjectBytes         *int64
}

// GridAccountDataSourceFilterModel describes the filter of the data source that lists the tenant accounts.
// StorageGRID has no generic query filter: the filter is applied on the list returned by the API.
type GridAccountDataSourceFilterModel struct {
        Name string `mapstructure:"name"`
}

// policyRequest builds the policy sent to StorageGRID.  current is the policy that StorageGRID has, if any.
// The settings that the provider does not manage are kept, as the API may replace the whole policy.
func (b GridAccountResourceBodyDataModel) policyRequest(current map[string]interface{}) map[string]interface{} {
        policy := make(map[string]interface{}, len(current)+6)
        for key, value := range current {
                policy[key] = value
        }
        policy["useAccountIdentitySource"] = b.Policy.UseAccountIdentitySource
        policy["allowPlatformServices"] = b.Policy.AllowPlatformServices
        if b.Policy.AllowSelectObjectContent != nil {
                policy["allowSelectObjectContent"] = *b.Policy.AllowSelectObjectContent
        }
        if b.Policy.AllowComplianceMode != nil {
                policy["allowComplianceMode"] = *b.Policy.AllowComplianceMode
        }
        if b.Policy.MaxRetentionDays != nil {
                policy["maxRetentionDays"] = *b.Policy.MaxRetentionDays
                if _, found := policy["maxRetentionYears"]; found {
                        // the maximum retention period has a value and a unit: days or years, not both
                        policy["maxRetentionYears"] = nil
                }
        }
        if b.Policy.QuotaObjectBytes != nil {
                policy["quotaObjectBytes"] = *b.Policy.QuotaObjectBytes
        } else {
                // null is no quota
                policy["quotaObjectBytes"] = nil
        }
        return policy
}

// request builds the body that is common to the creation and to the update of an account.
func (b GridAccountResourceBodyDataModel) request(currentPolicy map[string]interface{}) map[string]interface{} {
        capabilities := append([]string{}, b.Capabilities...)
        sort.Strings(capabilities)
        body := map[string]interface{}{
                "name":         b.Name,
                "capabilities": capabilities,
                "policy":       b.policyRequest(currentPolicy),
        }
        if b.Description != nil {
                body["description"] = *b.Description
        }
        return body
}

// createRequest builds the body of the creation of an account.
func (b GridAccountResourceBodyDataModel) createRequest() map[string]interface{} {
        body := b.request(nil)
        if b.Password != "" {
                body["password"] = b.Password
        }
        if b.GrantRootAccessToGroup != "" {
                body["grantRootAccessToGroup"] = b.GrantRootAccessToGroup
        }
        return body
}

// updateRequest builds the body of the update of an account.  current is the account that StorageGRID has.
//
// The PUT replaces the whole account, so everything that the provider does not manage is sent back as it is: the rules
// that synchronize the account with the other grids, the grid federation connections that the tenant can use, ...
// What StorageGRID assigns (the id), and what is only used when the account is created (the password and the group that
// has the root access), is not sent.
func (b GridAccountResourceBodyDataModel) updateRequest(current map[string]interface{}) map[string]interface{} {
        body := make(map[string]interface{}, len(current)+4)
        for key, value := range current {
                body[key] = value
        }
        for _, key := range []string{"id", "accountReplica", "password", "grantRootAccessToGroup"} {
                delete(body, key)
        }
        currentPolicy, _ := current["policy"].(map[string]interface{})
        for key, value := range b.request(currentPolicy) {
                body[key] = value
        }
        if b.Description == nil {
                // the description was removed
                delete(body, "description")
        }
        return body
}

// GetGridAccountByID to get grid_account info.  It returns nil, with no error, if the grid_account does not exist.
func GetGridAccountByID(errorHandler *utils.ErrorHandler, r *restclient.RestClient, id string) (*GridAccountGetDataModel, error) {
        api, err := gridAccountPath(errorHandler, id, "")
        if err != nil {
                return nil, err
        }
        statusCode, response, err := r.GetNilOrObject(api, nil)
        if err != nil {
                return nil, errorHandler.MakeAndReportError("error reading grid_account info", fmt.Sprintf("error on GET %s: %s, statusCode %d", api, err, statusCode))
        }
        if response == nil {
                return nil, nil
        }

        var data GridAccountGetDataModel
        if err := decode(response, &data); err != nil {
                return nil, errorHandler.MakeAndReportError(fmt.Sprintf("failed to decode response from GET %s", api),
                        fmt.Sprintf("error: %s, statusCode %d, response %#v", err, statusCode, response))
        }
        tflog.Debug(errorHandler.Ctx, fmt.Sprintf("Read grid_account: %#v", data))
        return &data, nil
}

// GetGridAccountByName to get grid_account info.  An error is reported unless exactly one grid_account has this name.
func GetGridAccountByName(errorHandler *utils.ErrorHandler, r *restclient.RestClient, name string) (*GridAccountGetDataModel, error) {
        found, err := GetAllGridAccount(errorHandler, r, &GridAccountDataSourceFilterModel{Name: name})
        if err != nil {
                return nil, err
        }
        switch len(found) {
        case 0:
                return nil, errorHandler.MakeAndReportError("grid_account not found", fmt.Sprintf("no grid_account with name %q", name))
        case 1:
                return &found[0], nil
        }
        return nil, errorHandler.MakeAndReportError("grid_account is not unique", fmt.Sprintf("%d grid_account have the name %q", len(found), name))
}

// GetAllGridAccount to get grid_account info for all the grid_accounts matching a filter
func GetAllGridAccount(errorHandler *utils.ErrorHandler, r *restclient.RestClient, filter *GridAccountDataSourceFilterModel) ([]GridAccountGetDataModel, error) {
        api := gridAccountsAPI
        // GetAllObjects follows the pagination of the API (limit and marker)
        statusCode, response, err := r.GetAllObjects(api, nil)
        if err != nil {
                return nil, errorHandler.MakeAndReportError("error reading grid_accounts info", fmt.Sprintf("error on GET %s: %s, statusCode %d", api, err, statusCode))
        }

        var data []GridAccountGetDataModel
        for _, info := range response {
                var record GridAccountGetDataModel
                if err := decode(info, &record); err != nil {
                        return nil, errorHandler.MakeAndReportError(fmt.Sprintf("failed to decode response from GET %s", api),
                                fmt.Sprintf("error: %s, statusCode %d, info %#v", err, statusCode, info))
                }
                if filter != nil && filter.Name != "" && record.Name != filter.Name {
                        continue
                }
                data = append(data, record)
        }
        tflog.Debug(errorHandler.Ctx, fmt.Sprintf("Read grid_accounts: %#v", data))
        return data, nil
}

// CreateGridAccount to create grid_account.  StorageGRID returns the new account, with its id.
func CreateGridAccount(errorHandler *utils.ErrorHandler, r *restclient.RestClient, body GridAccountResourceBodyDataModel) (*GridAccountGetDataModel, error) {
        api := gridAccountsAPI
        statusCode, response, err := r.CallCreateMethod(api, nil, body.createRequest())
        if err != nil {
                return nil, errorHandler.MakeAndReportError("error creating grid_account", fmt.Sprintf("error on POST %s: %s, statusCode %d", api, err, statusCode))
        }
        created, err := decodeGridAccountResponse(errorHandler, "POST", api, statusCode, response)
        if err != nil {
                return nil, err
        }
        if created.ID == "" {
                return nil, errorHandler.MakeAndReportError("error creating grid_account",
                        fmt.Sprintf("StorageGRID did not return the id of the account created by POST %s, statusCode %d: check if it exists", api, statusCode))
        }
        return created, nil
}

// UpdateGridAccount to update grid_account.  StorageGRID replaces the configuration of the account with the body of
// the PUT: the account is read first, and what the provider does not manage is sent back.
func UpdateGridAccount(errorHandler *utils.ErrorHandler, r *restclient.RestClient, id string, body GridAccountResourceBodyDataModel) (*GridAccountGetDataModel, error) {
        api, err := gridAccountPath(errorHandler, id, "")
        if err != nil {
                return nil, err
        }
        statusCode, current, err := r.GetNilOrObject(api, nil)
        if err != nil {
                return nil, errorHandler.MakeAndReportError("error reading grid_account before the update", fmt.Sprintf("error on GET %s: %s, statusCode %d", api, err, statusCode))
        }
        if current == nil {
                return nil, errorHandler.MakeAndReportError("grid_account not found", fmt.Sprintf("the grid_account %s does not exist anymore, run terraform plan to refresh the state", id))
        }

        statusCode, response, err := r.CallUpdateMethod(api, nil, body.updateRequest(current))
        if err != nil {
                return nil, errorHandler.MakeAndReportError("error updating grid_account", fmt.Sprintf("error on PUT %s: %s, statusCode %d", api, err, statusCode))
        }
        if _, ok := response.Object(); !ok {
                // the account is not in the response: read it
                updated, err := GetGridAccountByID(errorHandler, r, id)
                if err != nil {
                        return nil, err
                }
                if updated == nil {
                        return nil, errorHandler.MakeAndReportError("grid_account not found", fmt.Sprintf("the grid_account %s does not exist anymore, run terraform plan to refresh the state", id))
                }
                return updated, nil
        }
        return decodeGridAccountResponse(errorHandler, "PUT", api, statusCode, response)
}

// ChangeGridAccountPassword to change the password of the root user of the tenant of grid_account.
func ChangeGridAccountPassword(errorHandler *utils.ErrorHandler, r *restclient.RestClient, id string, password string) error {
        api, err := gridAccountPath(errorHandler, id, "/change-password")
        if err != nil {
                return err
        }
        statusCode, _, err := r.CallCreateMethod(api, nil, map[string]interface{}{"password": password})
        if err != nil {
                return errorHandler.MakeAndReportError("error changing the password of grid_account", fmt.Sprintf("error on POST %s: %s, statusCode %d", api, err, statusCode))
        }
        return nil
}

// DeleteGridAccount to delete grid_account.  It is not an error if the grid_account is already gone.
func DeleteGridAccount(errorHandler *utils.ErrorHandler, r *restclient.RestClient, id string) error {
        api, err := gridAccountPath(errorHandler, id, "")
        if err != nil {
                return err
        }
        statusCode, _, err := r.CallDeleteMethod(api, nil, nil)
        if err != nil && !restclient.IsNotFound(err) {
                return errorHandler.MakeAndReportError("error deleting grid_account", fmt.Sprintf("error on DELETE %s: %s, statusCode %d", api, err, statusCode))
        }
        return nil
}

// decodeGridAccountResponse reads the account returned by a POST or a PUT.
func decodeGridAccountResponse(errorHandler *utils.ErrorHandler, method string, api string, statusCode int, response restclient.RestResponse) (*GridAccountGetDataModel, error) {
        object, ok := response.Object()
        if !ok {
                return nil, errorHandler.MakeAndReportError(fmt.Sprintf("unexpected response from %s %s", method, api),
                        fmt.Sprintf("expected an object, statusCode %d, response %#v", statusCode, response.Data))
        }
        var data GridAccountGetDataModel
        if err := decode(object, &data); err != nil {
                return nil, errorHandler.MakeAndReportError(fmt.Sprintf("failed to decode response from %s %s", method, api),
                        fmt.Sprintf("error: %s, statusCode %d, response %#v", err, statusCode, object))
        }
        tflog.Debug(errorHandler.Ctx, fmt.Sprintf("%s %s: %#v", method, api, data))
        return &data, nil
}

// GridInfoSubsets returns all the subsets, sorted by name.
func GridInfoSubsets() []GridInfoSubset {
        subsets := append([]GridInfoSubset(nil), gridInfoSubsets...)
        sort.Slice(subsets, func(i, j int) bool { return subsets[i].Name < subsets[j].Name })

// GridInfoSubsetNames returns all the names that are accepted to select a subset, sorted: all, and the two names of each subset.
func GridInfoSubsetNames() []string {
        names := []string{GridInfoAll}
        for _, subset := range gridInfoSubsets {
                names = append(names, subset.Name, subset.InfoName)
        }
        for alias := range gridInfoAliases {
                names = append(names, alias)
        }
        sort.Strings(names)
        return names
}

// ResolveGridInfoSubsets returns the subsets that the names select, without duplicates, sorted by name.  All the subsets
// are selected by "all", and when there is no name.  An error lists the names that are not known.
func ResolveGridInfoSubsets(requested []string) ([]GridInfoSubset, error) {
        if len(requested) == 0 {
                requested = []string{GridInfoAll}
        }
        byName := make(map[string]GridInfoSubset, 2*len(gridInfoSubsets))
        for _, subset := range gridInfoSubsets {
                byName[subset.Name] = subset
                byName[subset.InfoName] = subset
        }
        for alias, name := range gridInfoAliases {
                byName[alias] = byName[name]
        }

        selected := map[string]GridInfoSubset{}
        var unknown []string
        for _, name := range requested {
                if name == GridInfoAll {
                        for _, subset := range gridInfoSubsets {
                                selected[subset.Name] = subset
                        }
                        continue
                }
                subset, found := byName[name]
                if !found {
                        unknown = append(unknown, name)
                        continue
                }
                selected[subset.Name] = subset
        }
        if len(unknown) > 0 {
                return nil, fmt.Errorf("unknown subsets: %s. A subset is %s, or the name of a REST API or an info name, for example grid/accounts or grid_accounts_info",
                        strings.Join(unknown, ", "), GridInfoAll)
        }
        subsets := make([]GridInfoSubset, 0, len(selected))
        for _, subset := range selected {
                subsets = append(subsets, subset)
        }
        sort.Slice(subsets, func(i, j int) bool { return subsets[i].Name < subsets[j].Name })
        return subsets, nil
}

// GridInfoResult is what was read for a subset.
type GridInfoResult struct {
        Subset GridInfoSubset
        // JSON is the data of the response, as JSON.  Sensitive values are masked, and the keys are sorted: it is the same
        // text whenever the data is the same.
        JSON       string
        StatusCode int
        Err        error
}

// IsSkippable returns true if the subset could not be read for a reason that is not a mistake of the request: the user
// is not allowed to read it, or this release of StorageGRID does not have it.
func (r GridInfoResult) IsSkippable() bool {
        var apiErr *restclient.APIError
        if !errors.As(r.Err, &apiErr) {
                return false
        }
        switch apiErr.StatusCode {
        case http.StatusForbidden, http.StatusNotFound, http.StatusMethodNotAllowed, http.StatusNotImplemented:
                return true
        }
        return false
}

// FetchGridInfo reads a subset.  parameters are added to the request, for instance limit.  The error is returned in the
// result and is not reported to Terraform, so that it can be called from several goroutines.
func FetchGridInfo(r *restclient.RestClient, subset GridInfoSubset, parameters map[string]string) GridInfoResult {
        result := GridInfoResult{Subset: subset}
        query := r.NewQuery()
        for name, value := range parameters {
                query.Set(name, value)
        }
        statusCode, data, err := r.GetData(subset.API, query)
        result.StatusCode = statusCode
        if err != nil {
                result.Err = err
                return result
        }
        // the encoder keeps < > & as they are, and the encoding of a map has its keys sorted
        var encoded bytes.Buffer
        encoder := json.NewEncoder(&encoded)
        encoder.SetEscapeHTML(false)
        if err := encoder.Encode(utils.RedactSensitive(data)); err != nil {
                result.Err = fmt.Errorf("unable to encode the response of GET %s: %w", subset.API, err)
                return result
        }
        result.JSON = strings.TrimSuffix(encoded.String(), "\n")
        return result
}

                return nil, errorHandler.MakeAndReportError("error reading grid product version", fmt.Sprintf("error on GET %s: %s, statusCode %d", api, err, statusCode))

        var data GridProductVersionGetDataModel
        if err := decode(response, &data); err != nil {
                return nil, errorHandler.MakeAndReportError(fmt.Sprintf("failed to decode response from GET %s", api),
                        fmt.Sprintf("error: %s, statusCode %d, response %#v", err, statusCode, response))
        }
        tflog.Debug(errorHandler.Ctx, fmt.Sprintf("Read grid_product_version data source: %#v", data))
        return &data, nil
}

// Profile describes how to reach and sign in to a grid, or to a tenant of a grid.
type Profile struct {
        Hostname  string
        Username  string
        Password  string
        AccountID string
        // ValidateCerts is true by default
        ValidateCerts         bool
        CACertFile            string
        APIVersion            string
        MaxConcurrentRequests int
}

// Config is created by the provider configure method.
// It is shared by all the resources and data sources, and must be used through a pointer.
type Config struct {
        ConnectionProfiles map[string]Profile
        Version            string

        mu      sync.Mutex
        clients map[string]*restclient.RestClient
}

// GetConnectionProfile retrieves a connection profile based on name
// If name is empty and only one profile is defined, it is returned
func (c *Config) GetConnectionProfile(name string) (string, *Profile, error) {
        if c == nil {
                return "", nil, fmt.Errorf("internal error, config is not initialized")
        }
        if len(c.ConnectionProfiles) == 0 {
                return "", nil, fmt.Errorf("error, at least one connection profile is required to connect to StorageGRID")
        }
        if name == "" && len(c.ConnectionProfiles) == 1 {
                for onlyName := range c.ConnectionProfiles {
                        name = onlyName
                }
        }
        if name == "" {
                names := make([]string, 0, len(c.ConnectionProfiles))
                for profileName := range c.ConnectionProfiles {
                        names = append(names, profileName)
                }
                sort.Strings(names)
                return "", nil, fmt.Errorf("error, connection profile name is required if more than one profile is defined: %s", strings.Join(names, ", "))
        }
        if profile, ok := c.ConnectionProfiles[name]; ok {
                return name, &profile, nil
        }
        return "", nil, fmt.Errorf("connection profile with name %s is not defined", name)
}

// NewClient returns the REST client of the connection profile identified by cxProfileName.
//
// There is a single client per profile.  It signs in once, and its token is used by every resource and data source
// that refers to the profile, instead of signing in for each call.  The returned client logs with the context of
// the error handler.
func (c *Config) NewClient(errorHandler *utils.ErrorHandler, cxProfileName string) (*restclient.RestClient, error) {
        name, profile, err := c.GetConnectionProfile(cxProfileName)
        if err != nil {
                return nil, errorHandler.MakeAndReportError("failed to set connection profile", err.Error())
        }

        c.mu.Lock()
        defer c.mu.Unlock()
        if client, ok := c.clients[name]; ok {
                return client.WithContext(errorHandler.Ctx), nil
        }

        // the tag is sent as the User-Agent, and used for telemetry
        tag := strings.Join([]string{"TerraformStorageGRID", c.Version}, "/")
        client, err := restclient.NewClient(errorHandler.Ctx, restclient.ConnectionProfile{
                Hostname:              profile.Hostname,
                Username:              profile.Username,
                Password:              profile.Password,
                AccountID:             profile.AccountID,
                ValidateCerts:         profile.ValidateCerts,
                CACertFile:            profile.CACertFile,
                APIVersion:            profile.APIVersion,
                MaxConcurrentRequests: profile.MaxConcurrentRequests,
        }, tag)
        if err != nil {
                return nil, errorHandler.MakeAndReportError("unable to create REST client",
                        fmt.Sprintf("error creating REST client for connection profile %s: %s", name, err))
        }
        if c.clients == nil {
                c.clients = map[string]*restclient.RestClient{}
        }
        c.clients[name] = client
        return client, nil
}

// Read refreshes the Terraform state with the latest data.
func (d *GridAccountDataSource) Read(ctx context.Context, req datasource.ReadRequest, resp *datasource.ReadResponse) {
        var data GridAccountDataSourceModel

        // Read Terraform configuration data into the model
        resp.Diagnostics.Append(req.Config.Get(ctx, &data)...)

        if resp.Diagnostics.HasError() {
                return
        }

        errorHandler := utils.NewErrorHandler(ctx, &resp.Diagnostics)
        // we need to defer setting the client until we can read the connection profile name
        client, err := connection.GetRestClient(errorHandler, d.config, data.CxProfileName)
        if err != nil {
                // error reporting done inside NewClient
                return
        }

        var account *interfaces.GridAccountGetDataModel
        if !data.ID.IsNull() && !data.ID.IsUnknown() {
                account, err = interfaces.GetGridAccountByID(errorHandler, client, data.ID.ValueString())
                if err != nil {
                        // error reporting done inside GetGridAccountByID
                        return
                }
                if account == nil {
                        errorHandler.MakeAndReportError("grid_account not found", fmt.Sprintf("no grid_account with id %s", data.ID.ValueString()))
                        return
                }
        } else {
                account, err = interfaces.GetGridAccountByName(errorHandler, client, data.Name.ValueString())
                if err != nil {
                        // error reporting done inside GetGridAccountByName
                        return
                }
        }

        item := newGridAccountDataSourceItem(ctx, account, &resp.Diagnostics)
        if resp.Diagnostics.HasError() {
                return
        }
        data.ID = item.ID
        data.Name = item.Name
        data.Description = item.Description
        data.Capabilities = item.Capabilities
        data.UseAccountIdentitySource = item.UseAccountIdentitySource
        data.AllowPlatformServices = item.AllowPlatformServices
        data.AllowSelectObjectContent = item.AllowSelectObjectContent
        data.AllowComplianceMode = item.AllowComplianceMode
        data.MaxRetentionDays = item.MaxRetentionDays
        data.QuotaObjectBytes = item.QuotaObjectBytes

        // Write logs using the tflog package
        // Documentation: https://terraform.io/plugin/log
        tflog.Debug(ctx, fmt.Sprintf("read a data source: %s", account.ID))

        // Save data into Terraform state
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
}

// exists, as StorageGRID only uses it when it creates the account.  The account is not replaced, as this destroys the
// tenant: use terraform apply -replace if this is what is wanted.
type rootAccessGroupIsSetOnce struct{}

// Description returns a plain text description of the modifier.
func (m rootAccessGroupIsSetOnce) Description(ctx context.Context) string {
        return "grant_root_access_to_group cannot be changed after the tenant account is created."
}

// MarkdownDescription returns a markdown description of the modifier.
func (m rootAccessGroupIsSetOnce) MarkdownDescription(ctx context.Context) string {
        return m.Description(ctx)
}

// PlanModifyString rejects a change from a value to another value.
func (m rootAccessGroupIsSetOnce) PlanModifyString(ctx context.Context, req planmodifier.StringRequest, resp *planmodifier.StringResponse) {
        if req.State.Raw.IsNull() || req.Plan.Raw.IsNull() {
                // create or destroy
                return
        }
        if err := checkRootAccessGroupIsNotChanged(req.StateValue, req.PlanValue); err != nil {
                resp.Diagnostics.AddAttributeError(req.Path, "grant_root_access_to_group cannot be changed", err.Error())
        }
}

// checkRootAccessGroupIsNotChanged returns an error if the group that was set is changed.  Setting it for the first
// time (an account that was imported) and removing it are accepted: StorageGRID is not told, as it cannot be.
func checkRootAccessGroupIsNotChanged(state types.String, plan types.String) error {
        if state.IsNull() || state.IsUnknown() || plan.IsNull() || plan.IsUnknown() || state.Equal(plan) {
                return nil
        }
        return fmt.Errorf("StorageGRID only uses the group that has the root access when it creates the tenant account, "+
                "and the group is %s: use terraform apply -replace to create the account again, "+
                "or change the permissions of the groups of the tenant", state.ValueString())
}

// toBody converts the model into what is sent to StorageGRID.  The password, and the group that has the root access,

// setFromAccount sets all the attributes that StorageGRID returns.  It is used to read the account: what the
// configuration says is replaced by what StorageGRID has.  The password and the group that has the root access are never
// returned, and are not changed.
func (m *GridAccountResourceModel) setFromAccount(ctx context.Context, account *interfaces.GridAccountGetDataModel, diags *diag.Diagnostics) {
        m.ID = types.StringValue(account.ID)
        m.Name = types.StringValue(account.Name)
        m.Description = types.StringPointerValue(account.Description)
        if account.Description != nil && *account.Description == "" {
                // an empty description is not a description
                m.Description = types.StringNull()
        }
        capabilities := account.Capabilities
        if capabilities == nil {
                capabilities = []string{}
        }
        set, setDiags := types.SetValueFrom(ctx, types.StringType, capabilities)
        diags.Append(setDiags...)
        m.Capabilities = set
        // when StorageGRID does not return a setting that always exists, the value that we know is kept
        if account.Policy.UseAccountIdentitySource != nil {
                m.UseAccountIdentitySource = types.BoolValue(*account.Policy.UseAccountIdentitySource)
        }
        if account.Policy.AllowPlatformServices != nil {
                m.AllowPlatformServices = types.BoolValue(*account.Policy.AllowPlatformServices)
        }
        m.AllowSelectObjectContent = types.BoolPointerValue(account.Policy.AllowSelectObjectContent)
        m.AllowComplianceMode = types.BoolPointerValue(account.Policy.AllowComplianceMode)
        m.MaxRetentionDays = types.Int64PointerValue(account.Policy.MaxRetentionDays)
        m.QuotaObjectBytes = types.Int64PointerValue(account.Policy.QuotaObjectBytes)
}

// setUnknownFromAccount sets the ID, and the attributes that are not known yet because they were not configured
// (StorageGRID chooses the value), from the account that StorageGRID returned for a create or an update.  The values that are
// known are the ones of the plan: Terraform requires them not to change, and the next refresh shows the differences.
func (m *GridAccountResourceModel) setUnknownFromAccount(account *interfaces.GridAccountGetDataModel) {
        m.ID = types.StringValue(account.ID)
        if m.AllowPlatformServices.IsUnknown() {
                m.AllowPlatformServices = types.BoolPointerValue(account.Policy.AllowPlatformServices)

        if m.AllowComplianceMode.IsUnknown() {

        }
        if m.MaxRetentionDays.IsUnknown() {
                m.MaxRetentionDays = types.Int64PointerValue(account.Policy.MaxRetentionDays)
        }
}

// hasUnknownSettings returns true if a setting that StorageGRID chooses when it is not configured is not known yet.

// settingsChanged returns true if the plan changes what is sent by an update of the account.
func (m *GridAccountResourceModel) settingsChanged(state *GridAccountResourceModel) bool {
        return !m.Name.Equal(state.Name) ||
                !m.Description.Equal(state.Description) ||
                !m.Capabilities.Equal(state.Capabilities) ||
                !m.UseAccountIdentitySource.Equal(state.UseAccountIdentitySource) ||
                !m.AllowPlatformServices.Equal(state.AllowPlatformServices) ||
                !m.AllowSelectObjectContent.Equal(state.AllowSelectObjectContent) ||
                !m.AllowComplianceMode.Equal(state.AllowComplianceMode) ||
                !m.MaxRetentionDays.Equal(state.MaxRetentionDays) ||
                !m.QuotaObjectBytes.Equal(state.QuotaObjectBytes)
}

// checkGridAccountVersion reports an error if the configuration has a setting that the version of StorageGRID
// does not have: StorageGRID would reject it with an error that does not tell which version is required.
func checkGridAccountVersion(errorHandler *utils.ErrorHandler, client *restclient.RestClient, m *GridAccountResourceModel) error {
        gates := []struct {
                attribute string
                set       bool
                major     int
                minor     int
        }{
                {"allow_select_object_content", optionalBool(m.AllowSelectObjectContent) != nil, 11, 6},
                {"allow_compliance_mode", optionalBool(m.AllowComplianceMode) != nil, 11, 9},
                {"max_retention_days", optionalInt64(m.MaxRetentionDays) != nil, 11, 9},
        }
        var version restclient.SGVersion
        versionRead := false
        for _, gate := range gates {
                if !gate.set {
                        continue
                }
                if !versionRead {
                        var err error
                        version, err = client.Version()
                        if err != nil {
                                return errorHandler.MakeAndReportError("error reading the StorageGRID version", err.Error())
                        }
                        versionRead = true
                }
                if !version.AtLeast(gate.major, gate.minor) {
                        return errorHandler.MakeAndReportError("unsupported attribute",
                                fmt.Sprintf("%s requires StorageGRID %d.%d or later, and this is StorageGRID %s.", gate.attribute, gate.major, gate.minor, version.Full))
                }
        }
        return nil
}

// Read refreshes the Terraform state with the latest data.
func (r *GridAccountResource) Read(ctx context.Context, req resource.ReadRequest, resp *resource.ReadResponse) {
        var data GridAccountResourceModel

        // Read Terraform prior state data into the model
        resp.Diagnostics.Append(req.State.Get(ctx, &data)...)

        if resp.Diagnostics.HasError() {
                return
        }

        errorHandler := utils.NewErrorHandler(ctx, &resp.Diagnostics)
        // we need to defer setting the client until we can read the connection profile name
        client, err := connection.GetRestClient(errorHandler, r.config, data.CxProfileName)
        if err != nil {
                // error reporting done inside NewClient
                return
        }

        account, err := interfaces.GetGridAccountByID(errorHandler, client, data.ID.ValueString())
        if err != nil {
                // error reporting done inside GetGridAccountByID
                return
        }
        if account == nil {
                // it was deleted outside of Terraform: remove it from the state, so that it is created again
                tflog.Warn(ctx, fmt.Sprintf("grid_account %s not found, removing it from the state", data.ID.ValueString()))
                resp.State.RemoveResource(ctx)
                return
        }

        data.setFromAccount(ctx, account, &resp.Diagnostics)
        if resp.Diagnostics.HasError() {
                return
        }

        // Write logs using the tflog package
        // Documentation: https://terraform.io/plugin/log
        tflog.Debug(ctx, fmt.Sprintf("read a resource: %s", account.ID))

        // Save data into Terraform state
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
}

// Create a resource and retrieve its ID
func (r *GridAccountResource) Create(ctx context.Context, req resource.CreateRequest, resp *resource.CreateResponse) {
        var data GridAccountResourceModel

        // Read Terraform plan data into the model
        resp.Diagnostics.Append(req.Plan.Get(ctx, &data)...)

        if resp.Diagnostics.HasError() {
                return
        }

        errorHandler := utils.NewErrorHandler(ctx, &resp.Diagnostics)
        client, err := connection.GetRestClient(errorHandler, r.config, data.CxProfileName)
        if err != nil {
                // error reporting done inside NewClient
                return
        }
        if err := checkGridAccountVersion(errorHandler, client, &data); err != nil {
                // error reporting done inside checkGridAccountVersion
                return
        }

        body := data.toBody(ctx, &resp.Diagnostics, true)
        if resp.Diagnostics.HasError() {
                return
        }
        created, err := interfaces.CreateGridAccount(errorHandler, client, body)
        if err != nil {
                // error reporting done inside CreateGridAccount
                return
        }

        account := created
        if data.hasUnknownSettings() {
                // StorageGRID may not return the settings that it chose when it creates the account: read them, so that the state
                // is the same as after a refresh.  The account exists, so a failure to read it does not fail the creation.
                if read, err := interfaces.GetGridAccountByID(utils.NewErrorHandler(ctx, &diag.Diagnostics{}), client, created.ID); err == nil && read != nil {
                        account = read
                }
        }
        data.setUnknownFromAccount(account)

        tflog.Trace(ctx, "created a resource")

        // Save data into Terraform state
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
}

// Update updates the resource and sets the updated Terraform state on success.
func (r *GridAccountResource) Update(ctx context.Context, req resource.UpdateRequest, resp *resource.UpdateResponse) {
        var plan, state GridAccountResourceModel

        // Read Terraform plan data and prior state data into the models
        resp.Diagnostics.Append(req.Plan.Get(ctx, &plan)...)
        resp.Diagnostics.Append(req.State.Get(ctx, &state)...)

        if resp.Diagnostics.HasError() {
                return
        }

        errorHandler := utils.NewErrorHandler(ctx, &resp.Diagnostics)
        client, err := connection.GetRestClient(errorHandler, r.config, plan.CxProfileName)
        if err != nil {
                // error reporting done inside NewClient
                return
        }
        if state.ID.IsNull() || state.ID.ValueString() == "" {
                errorHandler.MakeAndReportError("ID is missing", "grid_account ID is not set in the state")
                return
        }
        id := state.ID.ValueString()
        if err := checkRootAccessGroupIsNotChanged(state.GrantRootAccessToGroup, plan.GrantRootAccessToGroup); err != nil {
                errorHandler.MakeAndReportError("grant_root_access_to_group cannot be changed", err.Error())
                return
        }

        // what is saved in the state when the update fails: the password is only changed after the settings
        data := plan
        data.ID = state.ID
        data.Password = state.Password

        if plan.settingsChanged(&state) {
                if err := checkGridAccountVersion(errorHandler, client, &plan); err != nil {
                        // error reporting done inside checkGridAccountVersion
                        return
                }
                body := plan.toBody(ctx, &resp.Diagnostics, false)
                if resp.Diagnostics.HasError() {
                        return
                }
                updated, err := interfaces.UpdateGridAccount(errorHandler, client, id, body)
                if err != nil {
                        // error reporting done inside UpdateGridAccount
                        return
                }
                data.setUnknownFromAccount(updated)
        }
        // the settings are changed: save them now, so that they are not lost if the password cannot be changed
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
        if resp.Diagnostics.HasError() {
                return
        }

        if plan.Password.IsNull() || plan.Password.Equal(state.Password) {
                // removing the password from the configuration does not change the password of the tenant
                data.Password = plan.Password
        } else {
                if err := interfaces.ChangeGridAccountPassword(errorHandler, client, id, plan.Password.ValueString()); err != nil {
                        // error reporting done inside ChangeGridAccountPassword, the password is changed again by the next apply
                        return
                }
                data.Password = plan.Password
        }

        // Save updated data into Terraform state
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
}

// Read refreshes the Terraform state with the latest data.
func (d *GridInfoDataSource) Read(ctx context.Context, req datasource.ReadRequest, resp *datasource.ReadResponse) {
        var data GridInfoDataSourceModel

        // Read Terraform configuration data into the model
        resp.Diagnostics.Append(req.Config.Get(ctx, &data)...)

        if resp.Diagnostics.HasError() {
                return
        }

        var requested []string
        if !data.GatherSubset.IsNull() && !data.GatherSubset.IsUnknown() {
                resp.Diagnostics.Append(data.GatherSubset.ElementsAs(ctx, &requested, false)...)
        }
        parameters := map[string]string{}
        if !data.Parameters.IsNull() && !data.Parameters.IsUnknown() {
                resp.Diagnostics.Append(data.Parameters.ElementsAs(ctx, &parameters, false)...)
        }
        if resp.Diagnostics.HasError() {
                return
        }

        errorHandler := utils.NewErrorHandler(ctx, &resp.Diagnostics)
        subsets, err := interfaces.ResolveGridInfoSubsets(requested)
        if err != nil {
                errorHandler.MakeAndReportError("invalid gather_subset", err.Error())
                return
        }
        // with all, the subsets that cannot be read are skipped: the user may not be allowed to read them, or this release
        // of StorageGRID does not have them
        skipUnreadable := len(requested) == 0
        for _, name := range requested {
                skipUnreadable = skipUnreadable || name == interfaces.GridInfoAll
        }

        // we need to defer setting the client until we can read the connection profile name
        client, err := connection.GetRestClient(errorHandler, d.config, data.CxProfileName)
        if err != nil {
                // error reporting done inside NewClient
                return
        }

        // the subsets are read at the same time, the client limits the number of requests that are in progress
        results := make([]interfaces.GridInfoResult, len(subsets))
        var wait sync.WaitGroup
        for index := range subsets {
                wait.Add(1)
                go func(index int) {
                        defer wait.Done()
                        results[index] = interfaces.FetchGridInfo(client, subsets[index], parameters)
                }(index)
        }
        wait.Wait()

        info := make(map[string]string, len(results))
        var skipped []string
        for _, result := range results {
                switch {
                case result.Err == nil:
                        info[result.Subset.Name] = result.JSON
                case skipUnreadable && result.IsSkippable():
                        skipped = append(skipped, result.Subset.Name)
                        resp.Diagnostics.AddWarning(fmt.Sprintf("%s was skipped", result.Subset.Name),
                                fmt.Sprintf("GET %s: %s", result.Subset.API, result.Err))
                default:
                        errorHandler.MakeAndReportError(fmt.Sprintf("error reading %s", result.Subset.Name),
                                fmt.Sprintf("error on GET %s: %s, statusCode %d", result.Subset.API, result.Err, result.StatusCode))
                }
        }
        if resp.Diagnostics.HasError() {
                return
        }
        if len(info) == 0 {
                errorHandler.MakeAndReportError("no information", fmt.Sprintf("none of the %d subsets could be read", len(subsets)))
                return
        }

        infoMap, mapDiags := types.MapValueFrom(ctx, types.StringType, info)
        resp.Diagnostics.Append(mapDiags...)
        if resp.Diagnostics.HasError() {
                return
        }
        data.Info = infoMap

        // Write logs using the tflog package
        // Documentation: https://terraform.io/plugin/log
        tflog.Debug(ctx, fmt.Sprintf("read a data source: %d subsets, %d skipped", len(info), len(skipped)))

        // Save data into Terraform state
        resp.Diagnostics.Append(resp.State.Set(ctx, &data)...)
}

// Configure shared clients for data source and resource implementations.
func (p *StorageGRIDProvider) Configure(ctx context.Context, req provider.ConfigureRequest, resp *provider.ConfigureResponse) {
        var data StorageGRIDProviderModel

        resp.Diagnostics.Append(req.Config.Get(ctx, &data)...)
        if resp.Diagnostics.HasError() {
                tflog.Error(ctx, "unable to read the provider configuration")
                return
        }

        if data.ConnectionProfiles.IsUnknown() {
                resp.Diagnostics.AddError("no connection profiles", "At least one connection profile must be defined, and its value must be known.")
                return
        }
        if len(data.ConnectionProfiles.Elements()) == 0 {
                resp.Diagnostics.AddError("no connection profile", "At least one connection profile must be defined.")
                return
        }
        var profileModels []ConnectionProfileModel
        resp.Diagnostics.Append(data.ConnectionProfiles.ElementsAs(ctx, &profileModels, false)...)
        if resp.Diagnostics.HasError() {
                return
        }

        connectionProfiles := make(map[string]connection.Profile, len(profileModels))
        for _, model := range profileModels {
                name := model.Name.ValueString()
                if !requireKnownValue(&resp.Diagnostics, name, "name", model.Name) {
                        return
                }
                if _, found := connectionProfiles[name]; found {
                        resp.Diagnostics.AddError("Invalid configuration", fmt.Sprintf("connection profile '%s' is defined more than once.", name))
                        return
                }
                for attribute, value := range map[string]types.String{"hostname": model.Hostname, "username": model.Username, "password": model.Password} {
                        if !requireKnownValue(&resp.Diagnostics, name, attribute, value) {
                                return
                        }
                }
                for attribute, value := range map[string]types.String{"account_id": model.AccountID, "ca_cert_file": model.CACertFile, "api_version": model.APIVersion} {
                        if value.IsUnknown() {
                                resp.Diagnostics.AddError("Unknown value", fmt.Sprintf("%s of connection profile '%s' must be known when the provider is configured.", attribute, name))
                                return
                        }
                }
                if model.ValidateCerts.IsUnknown() {
                        resp.Diagnostics.AddError("Unknown value", fmt.Sprintf("validate_certs of connection profile '%s' must be known when the provider is configured.", name))
                        return
                }
                if !model.APIVersion.IsNull() && !apiVersionRegexp.MatchString(model.APIVersion.ValueString()) {
                        resp.Diagnostics.AddError("Invalid configuration",
                                fmt.Sprintf("api_version of connection profile '%s' must be a version like v3 or v4, got '%s'.", name, model.APIVersion.ValueString()))
                        return
                }

                validateCerts := true
                if !model.ValidateCerts.IsNull() {
                        validateCerts = model.ValidateCerts.ValueBool()
                }
                connectionProfiles[name] = connection.Profile{
                        Hostname:      model.Hostname.ValueString(),
                        Username:      model.Username.ValueString(),
                        Password:      model.Password.ValueString(),
                        AccountID:     model.AccountID.ValueString(),
                        ValidateCerts: validateCerts,
                        CACertFile:    model.CACertFile.ValueString(),
                        APIVersion:    model.APIVersion.ValueString(),
                }
        }

        config := &connection.Config{
                ConnectionProfiles: connectionProfiles,
                Version:            p.version,
        }
        resp.DataSourceData = config
        resp.ResourceData = config
}

// requireKnownValue reports an error unless value is set, and known.

        }
        if value.IsNull() || value.ValueString() == "" {
                diags.AddError("Missing required attribute", fmt.Sprintf("%s is required for connection profile '%s'.", attribute, profileName))
                return false
        }
        return true
}

package httpclient

import (
        "bytes"
        "encoding/json"
        "errors"
        "fmt"
        "net/http"
        "strings"

// maxRawBodyInError limits how much of a non JSON response is copied in an error.
const maxRawBodyInError = 300

// APIError is returned when StorageGRID answers with a status code outside of 2xx.
//
// StorageGRID reports errors as:
//
//        {"responseTime": "...", "status": "error", "apiVersion": "4.0", "code": 404,
//         "message": {"text": "...", "key": "..."}, "errors": [...]}
//
// The parsing is lenient: message may be a string or an object, errors may hold strings or objects,
// and the response may not be JSON at all (eg a 502 from a load balancer).
type APIError struct {
        StatusCode int
        Message    string
        Key        string
        Details    []string
        RawBody    string
}

// Error implements the error interface.
func (e *APIError) Error() string {
        var sb strings.Builder
        fmt.Fprintf(&sb, "StorageGRID returned statusCode %d", e.StatusCode)
        if e.Message != "" {
                fmt.Fprintf(&sb, ": %s", e.Message)
        }
        if e.Key != "" {
                fmt.Fprintf(&sb, " (key: %s)", e.Key)
        }
        if len(e.Details) > 0 {
                fmt.Fprintf(&sb, ", details: %s", strings.Join(e.Details, "; "))
        }
        if e.Message == "" && e.RawBody != "" {
                fmt.Fprintf(&sb, ", response: %s", e.RawBody)
        }
        return sb.String()
}

// IsNotFound returns true if err is an APIError for a 404.
func IsNotFound(err error) bool {
        var apiErr *APIError
        return errors.As(err, &apiErr) && apiErr.StatusCode == http.StatusNotFound
}

// ParseAPIError builds an APIError from an error response.
func ParseAPIError(statusCode int, body []byte) *APIError {
        apiErr := &APIError{StatusCode: statusCode}
        trimmed := bytes.TrimSpace(body)
        if len(trimmed) == 0 {
                return apiErr
        }

        var payload map[string]interface{}
        if err := json.Unmarshal(trimmed, &payload); err != nil {
                apiErr.RawBody = truncate(string(trimmed))
                return apiErr
        }

        switch message := payload["message"].(type) {
        case string:
                apiErr.Message = message
        case map[string]interface{}:
                apiErr.Message, _ = message["text"].(string)
                apiErr.Key, _ = message["key"].(string)
        }
        if details, ok := payload["errors"].([]interface{}); ok {
                for _, detail := range details {
                        apiErr.Details = append(apiErr.Details, describeDetail(detail))
                }
        }
        if apiErr.Message == "" && len(apiErr.Details) == 0 {
                apiErr.RawBody = truncate(string(trimmed))
        }
        return apiErr
}

func describeDetail(detail interface{}) string {
        switch typed := detail.(type) {
        case string:
                return typed
        case map[string]interface{}:
                if text, ok := typed["text"].(string); ok {
                        return text
                }
        }
        encoded, err := json.Marshal(detail)
        if err != nil {
                return fmt.Sprintf("%v", detail)
        }
        return string(encoded)
}

func truncate(s string) string {
        if len(s) > maxRawBodyInError {
                return s[:maxRawBodyInError] + "..."
        }
        return s
}

        apiRoot = "api"
        // authorizeAPI is served with v3 as it exists on every release, whatever the version used for the other calls.
        authorizeAPI   = "v3/authorize"
        defaultTimeout = 60 * time.Second
)

// HTTPProfile defines the connection attributes to build the base URL and to authenticate
type HTTPProfile struct {
        // Hostname of an admin node (or load balancer endpoint), with an optional port.  https:// is accepted.
        Hostname string
        Username string
        Password string
        // AccountID is the tenant account ID, to use the Tenant Management API.  Leave empty for the Grid Management API.
        AccountID     string
        ValidateCerts bool
        CACertFile    string
}

// HTTPClient represents a client for interaction with the StorageGRID management APIs.
// It signs in with the username and password, caches the bearer token, and signs in again
// when the token is rejected (tokens expire after 16 hours by default).
// It is safe for concurrent use.
type HTTPClient struct {
        cxProfile  HTTPProfile
        baseURL    *url.URL
        httpClient *http.Client
        tag        string

        tokenMu sync.Mutex
        token   string
}

// NewClient creates a new HTTP client.  No request is sent until the first call to Do.
func NewClient(cxProfile HTTPProfile, tag string) (*HTTPClient, error) {
        baseURL, err := parseBaseURL(cxProfile.Hostname)
        if err != nil {
                return nil, err
        }
        if cxProfile.Username == "" || cxProfile.Password == "" {
                return nil, errors.New("StorageGRID requires a username and a password")
        }
        tlsConfig := &tls.Config{
                MinVersion: tls.VersionTLS12,
                // #nosec G402 -- certificate validation is only skipped when the user sets validate_certs to false
                InsecureSkipVerify: !cxProfile.ValidateCerts,
        }
        if cxProfile.CACertFile != "" {
                caCert, err := os.ReadFile(cxProfile.CACertFile)
                if err != nil {
                        return nil, fmt.Errorf("failed to read CA certificate file: %w", err)
                }
                caCertPool := x509.NewCertPool()
                if !caCertPool.AppendCertsFromPEM(caCert) {
                        return nil, fmt.Errorf("failed to parse CA certificate file %s", cxProfile.CACertFile)
                }
                tlsConfig.RootCAs = caCertPool
        }
        return &HTTPClient{
                cxProfile: cxProfile,
                baseURL:   baseURL,
                httpClient: &http.Client{
                        Transport: &http.Transport{
                                Proxy:           http.ProxyFromEnvironment,
                                TLSClientConfig: tlsConfig,
                        },
                        Timeout: defaultTimeout,
                },
                tag: tag,
        }, nil
}

func parseBaseURL(hostname string) (*url.URL, error) {
        host := strings.TrimSpace(hostname)
        if host == "" {
                return nil, errors.New("StorageGRID hostname is required")
        }
        if !strings.Contains(host, "://") {
                host = "https://" + host
        }
        baseURL, err := url.Parse(host)
        if err != nil {
                return nil, fmt.Errorf("invalid StorageGRID hostname %q: %w", hostname, err)
        }
        if baseURL.Scheme != "https" {
                return nil, fmt.Errorf("invalid StorageGRID hostname %q: only https is supported", hostname)
        }
        if baseURL.Host == "" {
                return nil, fmt.Errorf("invalid StorageGRID hostname %q", hostname)
        }
        baseURL.Path = strings.TrimRight(baseURL.Path, "/")
        return baseURL, nil
}

// buildURL builds https://<host>/api/<path>.  path is relative to /api, eg v4/grid/accounts
func (c *HTTPClient) buildURL(path string) string {
        return c.baseURL.JoinPath(apiRoot, path).String()
}

// Do sends the API request, and returns the HTTP status code and the response body.
// path is relative to the API root and starts with the API version, eg v4/grid/accounts
//
// An error is only returned if the request could not be sent, or if the sign in failed.
// A response with a status code >= 300 is not an error at this level.
func (c *HTTPClient) Do(ctx context.Context, path string, req *Request) (int, []byte, error) {
        token, err := c.getToken(ctx)
        if err != nil {
                return -1, nil, err
        }
        statusCode, body, err := c.send(ctx, path, req, token)
        if err == nil && statusCode == http.StatusUnauthorized {
                // the token has expired or was revoked: sign in again, and retry once
                tflog.Debug(ctx, "token was rejected, signing in again")
                c.invalidateToken(token)
                token, err = c.getToken(ctx)
                if err != nil {
                        return -1, nil, err
                }
                statusCode, body, err = c.send(ctx, path, req, token)
        }
        return statusCode, body, err
}

func (c *HTTPClient) send(ctx context.Context, path string, req *Request, token string) (int, []byte, error) {
        httpReq, err := req.buildHTTPReq(c.buildURL(path))
        if err != nil {
                return -1, nil, err
        }
        httpReq = httpReq.WithContext(ctx)
        httpReq.Header.Set("Authorization", "Bearer "+token)
        return c.roundTrip(ctx, httpReq)
}

func (c *HTTPClient) roundTrip(ctx context.Context, httpReq *http.Request) (int, []byte, error) {
        httpReq.Header.Set("Accept", "application/json")
        httpReq.Header.Set("Cache-Control", "no-cache")
        if httpReq.Body != nil {
                httpReq.Header.Set("Content-Type", "application/json")
        }
        httpReq.Header.Set("User-Agent", c.tag)

        tflog.Debug(ctx, fmt.Sprintf("sending: %s %s", httpReq.Method, httpReq.URL.String()))
        httpRes, err := c.httpClient.Do(httpReq)
        if err != nil {
                tflog.Error(ctx, fmt.Sprintf("HTTP request failed: %s %s: %s", httpReq.Method, httpReq.URL.String(), err))
                return -1, nil, err
        }
        defer httpRes.Body.Close()

        body, err := io.ReadAll(httpRes.Body)
        if err != nil {
                tflog.Error(ctx, fmt.Sprintf("HTTP response read failed: %s, statusCode: %d", err, httpRes.StatusCode))
                return httpRes.StatusCode, nil, err
        }
        tflog.Debug(ctx, fmt.Sprintf("received: %s %s %d", httpReq.Method, httpReq.URL.String(), httpRes.StatusCode))
        return httpRes.StatusCode, body, nil
}

// getToken returns the cached token, signing in first if needed.
func (c *HTTPClient) getToken(ctx context.Context) (string, error) {
        c.tokenMu.Lock()
        defer c.tokenMu.Unlock()
        if c.token != "" {
                return c.token, nil
        }
        token, err := c.authorize(ctx)
        if err != nil {
                return "", err
        }
        c.token = token
        return token, nil
}

// invalidateToken clears the cached token, unless another goroutine already replaced it.
func (c *HTTPClient) invalidateToken(rejected string) {
        c.tokenMu.Lock()
        defer c.tokenMu.Unlock()
        if c.token == rejected {
                c.token = ""
        }
}

// authorize signs in and returns a bearer token.
// The body is deliberately never logged, as it holds the password.
func (c *HTTPClient) authorize(ctx context.Context) (string, error) {
        payload := map[string]interface{}{
                "username":  c.cxProfile.Username,
                "password":  c.cxProfile.Password,
                "cookie":    false,
                "csrfToken": false,
        }
        if c.cxProfile.AccountID != "" {
                payload["accountId"] = c.cxProfile.AccountID
        }
        body, err := json.Marshal(payload)
        if err != nil {
                return "", err
        }
        httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, c.buildURL(authorizeAPI), bytes.NewReader(body))
        if err != nil {
                return "", err
        }
        statusCode, response, err := c.roundTrip(ctx, httpReq)
        if err != nil {
                return "", fmt.Errorf("unable to sign in to StorageGRID: %w", err)
        }
        if statusCode < 200 || statusCode >= 300 {
                return "", fmt.Errorf("unable to sign in to StorageGRID: %w", ParseAPIError(statusCode, response))
        }
        var result struct {
                Data string `json:"data"`
        }
        if err := json.Unmarshal(response, &result); err != nil || result.Data == "" {
                return "", fmt.Errorf("unable to sign in to StorageGRID: no token in the response, statusCode %d", statusCode)
        }
        return result.Data, nil
}

// Request represents a request to a REST API
type Request struct {
        Method string
        // Body is marshalled as JSON.  nil means no body.
        // It is an interface{} and not a map because some StorageGRID APIs expect a JSON array.
        Body  interface{}
        Query url.Values
}

// hasBody returns false for nil and for typed nil values (eg a nil map).
func hasBody(body interface{}) bool {
        if body == nil {
                return false
        }
        value := reflect.ValueOf(body)
        switch value.Kind() {
        case reflect.Map, reflect.Slice, reflect.Ptr, reflect.Interface:
                return !value.IsNil()
        }
        return true
}

// buildHTTPReq builds an HTTP request, without authentication.
func (r *Request) buildHTTPReq(requestURL string) (*http.Request, error) {
        var body io.Reader
        if hasBody(r.Body) {
                bodyJSON, err := json.Marshal(r.Body)
                if err != nil {
                        return nil, err
                }
                body = bytes.NewReader(bodyJSON)
        }
        req, err := http.NewRequest(r.Method, requestURL, body)
        if err != nil {
                return nil, err
        }
        if len(r.Query) != 0 {
                req.URL.RawQuery = r.Query.Encode()
        }
        return req, nil
}

const (

        // pageSize is the value of the limit parameter used to read paginated lists.
        pageSize = 20
        // probeAPIVersion is used to read the product version, before we know which version to use.
        probeAPIVersion = "v3"

// ConnectionProfile describes how to reach and sign in to a grid, or to a tenant of a grid.
type ConnectionProfile struct {
        Hostname string
        Username string
        Password string
        // AccountID is the ID of a tenant account.  When set, the client uses the Tenant Management API (/org),
        // otherwise it uses the Grid Management API (/grid).
        AccountID     string
        ValidateCerts bool
        CACertFile    string
        // APIVersion forces the API version, eg v4.  By default it is selected from the StorageGRID version.
        APIVersion            string
        MaxConcurrentRequests int

// RestClient to interact with the StorageGRID REST APIs.
//
// All the state that is worth sharing (token, API version, request throttling) lives in a session shared by
// every copy of the client.  A RestClient only adds the context used for logging: use WithContext to get one for
// the current Terraform operation.
type RestClient struct {
        ctx context.Context
        s   *session
}

type session struct {
        connectionProfile ConnectionProfile
        httpClient        *httpclient.HTTPClient
        requestSlots      chan struct{}

        versionMu  sync.Mutex
        apiVersion string
        sgVersion  *SGVersion
}

// NewClient creates a new REST client.  The client signs in when it sends its first request.
// tag is sent as the User-Agent, for telemetry.
func NewClient(ctx context.Context, cxProfile ConnectionProfile, tag string) (*RestClient, error) {
        httpClient, err := httpclient.NewClient(httpclient.HTTPProfile{
                Hostname:      cxProfile.Hostname,
                Username:      cxProfile.Username,
                Password:      cxProfile.Password,
                AccountID:     cxProfile.AccountID,
                ValidateCerts: cxProfile.ValidateCerts,
                CACertFile:    cxProfile.CACertFile,
        }, tag)
        if err != nil {
                return nil, err
        }
        maxConcurrentRequests := cxProfile.MaxConcurrentRequests
        if maxConcurrentRequests <= 0 {
                maxConcurrentRequests = defaultMaxConcurrentRequests
        }
        return &RestClient{
                ctx: ctx,
                s: &session{
                        connectionProfile: cxProfile,
                        httpClient:        httpClient,
                        requestSlots:      make(chan struct{}, maxConcurrentRequests),
                },
        }, nil
}

// WithContext returns a client that shares the session of r, and logs with ctx.
func (r *RestClient) WithContext(ctx context.Context) *RestClient {
        return &RestClient{ctx: ctx, s: r.s}
}

// IsTenant returns true if the client uses the Tenant Management API.
func (r *RestClient) IsTenant() bool {
        return r.s.connectionProfile.AccountID != ""
}

// apiRoot returns grid or org: the first element of the path of the configuration APIs.
func (r *RestClient) apiRoot() string {
        if r.IsTenant() {
                return "org"
        }
        return "grid"
}

// Version returns the StorageGRID software version, which is read the first time it is needed.
// Use it to check that a feature is supported: client.Version() then version.AtLeast(11, 9)
func (r *RestClient) Version() (SGVersion, error) {
        r.s.versionMu.Lock()
        defer r.s.versionMu.Unlock()
        if err := r.resolveSGVersionLocked(); err != nil {
                return SGVersion{}, err
        }
        return *r.s.sgVersion, nil
}

// APIVersion returns the API version used in the requests, eg v4.
func (r *RestClient) APIVersion() (string, error) {
        r.s.versionMu.Lock()
        defer r.s.versionMu.Unlock()
        if r.s.apiVersion != "" {
                return r.s.apiVersion, nil
        }
        if forced := NormalizeAPIVersion(r.s.connectionProfile.APIVersion); forced != "" {
                r.s.apiVersion = forced
                return forced, nil
        }
        if err := r.resolveSGVersionLocked(); err != nil {
                return "", err
        }
        r.s.apiVersion = apiVersionFor(*r.s.sgVersion)
        tflog.Debug(r.ctx, fmt.Sprintf("StorageGRID %s, using API %s", r.s.sgVersion.Full, r.s.apiVersion))
        return r.s.apiVersion, nil
}

// resolveSGVersionLocked reads the product version.  versionMu must be held.
func (r *RestClient) resolveSGVersionLocked() error {
        if r.s.sgVersion != nil {
                return nil
        }
        path := probeAPIVersion + "/" + r.apiRoot() + "/config/product-version"
        statusCode, body, err := r.s.httpClient.Do(r.ctx, path, &httpclient.Request{Method: "GET"})
        response, err := unmarshalResponse(statusCode, body, err)
        if err != nil {
                return fmt.Errorf("unable to read the StorageGRID version from GET %s: %w", path, err)
        }
        object, ok := response.Object()
        if !ok {
                return fmt.Errorf("unable to read the StorageGRID version from GET %s: %w", path, errUnexpectedData)
        }
        full, _ := object["productVersion"].(string)
        version, err := ParseSGVersion(full)
        if err != nil {
                return err
        }
        r.s.sgVersion = &version
        return nil
}

// NewQuery is used to provide query parameters.  Set and Add functions are inherited from url.Values
func (r *RestClient) NewQuery() *RestQuery {
        return &RestQuery{Values: url.Values{}}
}

// RestQuery is a wrapper around url.Values.
type RestQuery struct {
        url.Values
}

// SetValues adds a set of key, value.  Empty values are ignored.
func (q *RestQuery) SetValues(keyValues map[string]interface{}) {
        for k, v := range keyValues {
                value := fmt.Sprintf("%v", v)
                if value != "" {
                        q.Set(k, value)
                }
        }
}

func (q *RestQuery) clone() *RestQuery {
        clone := &RestQuery{Values: url.Values{}}
        if q != nil {
                for k, v := range q.Values {
                        clone.Values[k] = append([]string(nil), v...)
                }
        }
        return clone
}

// callAPIMethod can be used to make a request to any REST API method.
// path is relative to the API version, eg grid/accounts
func (r *RestClient) callAPIMethod(method string, path string, query *RestQuery, body interface{}) (int, RestResponse, error) {
        apiVersion, err := r.APIVersion()
        if err != nil {
                return -1, RestResponse{}, err
        }

        r.s.requestSlots <- struct{}{}
        defer func() { <-r.s.requestSlots }()

        values := url.Values{}
        if query != nil {
                values = query.Values
        }
        fullPath := apiVersion + "/" + strings.TrimPrefix(path, "/")
        if body != nil {
                tflog.Debug(r.ctx, fmt.Sprintf("request body for %s %s", method, fullPath), map[string]interface{}{"body": utils.RedactSensitive(body)})
        }
        statusCode, rawBody, httpClientErr := r.s.httpClient.Do(r.ctx, fullPath, &httpclient.Request{
                Method: method,
                Body:   body,
                Query:  values,
        })
        response, err := unmarshalResponse(statusCode, rawBody, httpClientErr)
        if err != nil {
                tflog.Debug(r.ctx, fmt.Sprintf("%s %s failed, statusCode %d: %s", method, fullPath, statusCode, err))
                return statusCode, response, err
        }
        tflog.Debug(r.ctx, fmt.Sprintf("response data for %s %s", method, fullPath), map[string]interface{}{"data": utils.RedactSensitive(response.Data)})
        return statusCode, response, nil
}

// CallCreateMethod returns response from POST results.  An error is reported if an error is received.
func (r *RestClient) CallCreateMethod(path string, query *RestQuery, body interface{}) (int, RestResponse, error) {
        return r.callAPIMethod("POST", path, query, body)
}

func (r *RestClient) CallUpdateMethod(path string, query *RestQuery, body interface{}) (int, RestResponse, error) {

// GetData returns the data of a single GET, whatever its type: an object, a list of objects, a list of strings, a
// string.  It is nil when the response has no data.  Use it for the APIs that do not return an object.
func (r *RestClient) GetData(path string, query *RestQuery) (int, interface{}, error) {
        statusCode, response, err := r.callAPIMethod("GET", path, query, nil)
        if err != nil {
                return statusCode, nil, err
        }
        return statusCode, response.Data, nil
}

// GetNilOrObject returns nil if the resource is not found (404) or a single object.
func (r *RestClient) GetNilOrObject(path string, query *RestQuery) (int, map[string]interface{}, error) {
        statusCode, response, err := r.callAPIMethod("GET", path, query, nil)
        if err != nil {
                if IsNotFound(err) {
                        return statusCode, nil, nil
                }
                return statusCode, nil, err
        }
        object, ok := response.Object()
        if !ok {
                return statusCode, nil, fmt.Errorf("GET %s: %w, expected an object, got %T", path, errUnexpectedData, response.Data)
        }
        return statusCode, object, nil
}

// GetZeroOrMoreObjects returns the list of objects of a single GET.
// StorageGRID may split large lists in pages, use GetAllObjects for the APIs that support pagination.
func (r *RestClient) GetZeroOrMoreObjects(path string, query *RestQuery) (int, []map[string]interface{}, error) {
        statusCode, response, err := r.callAPIMethod("GET", path, query, nil)
        if err != nil {
                return statusCode, nil, err
        }
        objects, ok := response.List()
        if !ok {
                return statusCode, nil, fmt.Errorf("GET %s: %w, expected a list of objects, got %T", path, errUnexpectedData, response.Data)
        }
        return statusCode, objects, nil
}

// GetAllObjects returns all the objects of an API that supports pagination (limit and marker parameters),
// like grid/accounts.  The marker is the id of the last object of the previous page.
func (r *RestClient) GetAllObjects(path string, query *RestQuery) (int, []map[string]interface{}, error) {
        pageQuery := query.clone()
        pageQuery.Set("limit", fmt.Sprintf("%d", pageSize))
        var all []map[string]interface{}
        statusCode := -1
        for {
                var page []map[string]interface{}
                var err error
                statusCode, page, err = r.GetZeroOrMoreObjects(path, pageQuery)
                if err != nil {
                        return statusCode, nil, err
                }
                if len(page) == 0 {
                        return statusCode, all, nil
                }
                last, ok := page[len(page)-1]["id"]
                if !ok {
                        return statusCode, nil, fmt.Errorf("GET %s: unable to read the next page, the last object has no id", path)
                }
                marker := fmt.Sprintf("%v", last)
                if marker == pageQuery.Get("marker") {
                        // the API ignores the marker, and does not support pagination: this page is the one we already have
                        return statusCode, all, nil
                }
                all = append(all, page...)
                pageQuery.Set("marker", marker)
        }
}

// APIError is the error returned when StorageGRID answers with a status code outside of 2xx.
type APIError = httpclient.APIError

// IsNotFound returns true if err is an APIError for a 404.
func IsNotFound(err error) bool {
        return httpclient.IsNotFound(err)
}

// RestResponse is the success envelope used by all StorageGRID management APIs:
//
//        {"responseTime": "2024-01-01T00:00:00.000Z", "status": "success", "apiVersion": "4.0", "data": ...}
//
// data is an object for a single resource, and an array for a list.  It is absent when the status code is 204.
type RestResponse struct {
        StatusCode   int
        Status       string
        APIVersion   string
        ResponseTime string
        Data         interface{}
}

// Object returns data as an object.
func (r RestResponse) Object() (map[string]interface{}, bool) {
        object, ok := r.Data.(map[string]interface{})
        return object, ok
}

// List returns data as a list of objects.  A response without data is an empty list.
func (r RestResponse) List() ([]map[string]interface{}, bool) {
        if r.Data == nil {
                return []map[string]interface{}{}, true
        }
        items, ok := r.Data.([]interface{})
        if !ok {
                return nil, false
        }
        objects := make([]map[string]interface{}, 0, len(items))
        for _, item := range items {
                object, ok := item.(map[string]interface{})
                if !ok {
                        return nil, false
                }
                objects = append(objects, object)
        }
        return objects, true
}

// unmarshalResponse converts the HTTP response into a RestResponse, or into an APIError if the call failed.
func unmarshalResponse(statusCode int, body []byte, httpClientErr error) (RestResponse, error) {
        response := RestResponse{StatusCode: statusCode}
        if httpClientErr != nil {
                return response, httpClientErr
        }
        if statusCode < 200 || statusCode >= 300 {
                return response, httpclient.ParseAPIError(statusCode, body)
        }
        if len(bytes.TrimSpace(body)) == 0 {
                // 204 No Content, as returned by most DELETE
                return response, nil
        }
        var envelope struct {
                ResponseTime string      `json:"responseTime"`
                Status       string      `json:"status"`
                APIVersion   string      `json:"apiVersion"`
                Data         interface{} `json:"data"`
        }
        // UseNumber keeps large integers (quotas, counters) exact, instead of converting them to float64
        decoder := json.NewDecoder(bytes.NewReader(body))
        decoder.UseNumber()
        if err := decoder.Decode(&envelope); err != nil {
                return response, fmt.Errorf("unable to decode the response, statusCode %d: %w", statusCode, err)
        }
        response.ResponseTime = envelope.ResponseTime
        response.Status = envelope.Status
        response.APIVersion = envelope.APIVersion
        response.Data = envelope.Data
        return response, nil
}

import (
        "fmt"
        "strconv"
        "strings"
        "unicode"
)

// SGVersion is the StorageGRID software version.
type SGVersion struct {
        Major int
        Minor int
        // Full is the version as reported by the product-version API, eg 11.9.0
        Full string
}

// AtLeast returns true if the version is equal to or newer than major.minor.
func (v SGVersion) AtLeast(major int, minor int) bool {
        if v.Major != major {
                return v.Major > major
        }
        return v.Minor >= minor
}

// ParseSGVersion parses the productVersion reported by StorageGRID, eg 11.9.0 or 12.0.0-20250101.1234.abcdef
func ParseSGVersion(full string) (SGVersion, error) {
        parts := strings.SplitN(strings.TrimSpace(full), ".", 3)
        if len(parts) < 2 {
                return SGVersion{}, fmt.Errorf("unable to parse StorageGRID version %q", full)
        }
        major, err := strconv.Atoi(leadingDigits(parts[0]))
        if err != nil {
                return SGVersion{}, fmt.Errorf("unable to parse StorageGRID version %q", full)
        }
        minor, err := strconv.Atoi(leadingDigits(parts[1]))
        if err != nil {
                return SGVersion{}, fmt.Errorf("unable to parse StorageGRID version %q", full)
        }
        return SGVersion{Major: major, Minor: minor, Full: strings.TrimSpace(full)}, nil
}

func leadingDigits(s string) string {
        end := 0
        for end < len(s) && unicode.IsDigit(rune(s[end])) {
                end++
        }
        return s[:end]
}

// NormalizeAPIVersion accepts 4, v4, or V4 and returns v4.
func NormalizeAPIVersion(version string) string {
        version = strings.ToLower(strings.TrimSpace(version))
        if version == "" {
                return ""
        }
        if !strings.HasPrefix(version, "v") {
                version = "v" + version
        }
        return version
}

// apiVersionFor returns the API version to use for a given software version.
// v4 was introduced with StorageGRID 11.8.
func apiVersionFor(version SGVersion) string {
        if version.AtLeast(11, 8) {
                return "v4"
        }
        return "v3"
}

import (
        "context"
        "errors"
        "fmt"

        "github.com/hashicorp/terraform-plugin-framework/diag"
        "github.com/hashicorp/terraform-plugin-log/tflog"
)

// ErrorHandler creates an error handler to combine logging and reporting errors
type ErrorHandler struct {
        Ctx    context.Context
        diags  *diag.Diagnostics
        name   string
        subCtx context.Context
}

// NewErrorHandler creates an error handler based on current context and TF diagnostics
func NewErrorHandler(ctx context.Context, diags *diag.Diagnostics) *ErrorHandler {
        name := "error_handler"
        return &ErrorHandler{
                Ctx:    ctx,
                diags:  diags,
                name:   name,
                subCtx: tflog.NewSubsystem(ctx, name, tflog.WithAdditionalLocationOffset(1)),
        }
}

// MakeAndLogError builds an error using message and logs the error with tflog
func (e *ErrorHandler) MakeAndLogError(msg string) error {
        e.validate()
        tflog.WithAdditionalLocationOffset(1)
        tflog.SubsystemError(e.subCtx, e.name, msg)
        return errors.New(msg)
}

// MakeAndReportError builds an error using message and logs the error with tflog
// The error is added to the diagnostic and will be reported by Terraform
func (e *ErrorHandler) MakeAndReportError(summary string, msg string) error {
        e.validate()
        fullMsg := fmt.Sprintf("%s: %s", summary, msg)
        tflog.SubsystemError(e.subCtx, e.name, msg)
        e.diags.AddError(summary, msg)
        return errors.New(fullMsg)
}

func (e *ErrorHandler) validate() {
        if e == nil {
                panic("Error handler is not set")
        }
}

package utils

import "strings"

const redactedValue = "********"

// sensitiveKeyParts are matched (case insensitive) against JSON keys.
// StorageGRID bodies carry tenant root passwords, S3 secret keys, tokens, SNMP community strings, ...
var sensitiveKeyParts = []string{"password", "passphrase", "secret", "token", "privatekey", "credential", "community"}

func isSensitiveKey(key string) bool {
        lower := strings.ToLower(key)
        for _, part := range sensitiveKeyParts {
                if strings.Contains(lower, part) {
                        return true
                }
        }
        return false
}

// RedactSensitive returns a copy of a decoded JSON value, with the value of every sensitive key masked.
// It must be used before logging request or response bodies.
func RedactSensitive(v interface{}) interface{} {
        switch typed := v.(type) {
        case map[string]interface{}:
                out := make(map[string]interface{}, len(typed))
                for key, value := range typed {
                        if isSensitiveKey(key) {
                                out[key] = redactedValue
                        } else {
                                out[key] = RedactSensitive(value)
                        }
                }
                return out
        case []interface{}:
                out := make([]interface{}, len(typed))
                for index, value := range typed {
                        out[index] = RedactSensitive(value)
                }
                return out
        case []map[string]interface{}:
                out := make([]interface{}, len(typed))
                for index, value := range typed {
                        out[index] = RedactSensitive(value)
                }
                return out
        default:
                return v
        }
}
